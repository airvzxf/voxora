1//! `.voxora-manifest.json` — on-disk metadata for a cached model.
2//!
3//! Written next to the cached weights so a registry can answer
4//! "which engine does this directory belong to?" without re-parsing
5//! every file in the directory.
6
7use std::path::{Path, PathBuf};
8
9use serde::{Deserialize, Serialize};
10
11use voxora_engine::EngineFamily;
12
13use crate::error::RegistryError;
14
15/// Filename of the on-disk manifest written next to cached weights.
16pub const MANIFEST_FILENAME: &str = ".voxora-manifest.json";
17/// Bump this whenever the manifest schema changes in a
18/// non-backwards-compatible way.
19pub const MANIFEST_VERSION: u32 = 1;
20
21/// On-disk sidecar that records which engine a cached model directory
22/// belongs to. Written by [`CacheManifest::write`], read by
23/// [`CacheManifest::read`].
24#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
25#[non_exhaustive]
26pub struct CacheManifest {
27    /// Schema version this manifest was written with.
28    pub manifest_version: u32,
29    /// Engine the cached weights belong to. Serialised via
30    /// `family_serde` so we don't depend on `voxora-engine` having a
31    /// `serde` feature.
32    #[serde(with = "family_serde")]
33    pub family: EngineFamily,
34    /// Stable identifier (HF `org/repo[/file]` or local absolute path).
35    pub model_id: String,
36    /// When the manifest was last written (seconds since UNIX epoch).
37    pub written_at_unix: u64,
38}
39
40impl CacheManifest {
41    /// Build a [`CacheManifest`] with the current schema version.
42    pub fn new(family: EngineFamily, model_id: impl Into<String>, written_at_unix: u64) -> Self {
43        Self {
44            manifest_version: MANIFEST_VERSION,
45            family,
46            model_id: model_id.into(),
47            written_at_unix,
48        }
49    }
50
51    /// Read a manifest at `dir/.voxora-manifest.json`.
52    pub fn read(dir: &Path) -> Result<Self, RegistryError> {
53        let p = dir.join(MANIFEST_FILENAME);
54        let text = std::fs::read_to_string(&p)
55            .map_err(|e| RegistryError::Parse(format!("manifest read {p:?}: {e}")))?;
56        serde_json::from_str(&text)
57            .map_err(|e| RegistryError::Parse(format!("manifest parse {p:?}: {e}")))
58    }
59
60    /// Write this manifest to `dir/.voxora-manifest.json`.
61    pub fn write(&self, dir: &Path) -> Result<PathBuf, RegistryError> {
62        let p = dir.join(MANIFEST_FILENAME);
63        let text = serde_json::to_string_pretty(self)
64            .map_err(|e| RegistryError::Parse(format!("manifest serialize: {e}")))?;
65        std::fs::write(&p, text)
66            .map_err(|e| RegistryError::Parse(format!("manifest write {p:?}: {e}")))?;
67        Ok(p)
68    }
69}
70
71mod family_serde {
72    //! `EngineFamily` doesn't derive `Serialize`/`Deserialize`, so we
73    //! round-trip it through its canonical config spelling.
74    use serde::{Deserialize, Deserializer, Serialize, Serializer};
75    use voxora_engine::EngineFamily;
76
77    pub fn serialize<S: Serializer>(family: &EngineFamily, s: S) -> Result<S::Ok, S::Error> {
78        family.as_config().serialize(s)
79    }
80
81    pub fn deserialize<'de, D: Deserializer<'de>>(d: D) -> Result<EngineFamily, D::Error> {
82        let raw = String::deserialize(d)?;
83        EngineFamily::from_config(&raw)
84            .ok_or_else(|| serde::de::Error::custom(format!("unknown engine family {raw:?}")))
85    }
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91
92    #[test]
93    fn round_trip_via_disk() {
94        let dir = tempfile::tempdir().unwrap();
95        let m = CacheManifest::new(
96            voxora_engine::EngineFamily::Whisper,
97            "ggerganov/whisper.cpp",
98            1_700_000_000,
99        );
100        m.write(dir.path()).expect("write");
101        let back = CacheManifest::read(dir.path()).expect("read");
102        assert_eq!(m, back);
103    }
104
105    #[test]
106    fn read_missing_returns_error() {
107        let dir = tempfile::tempdir().unwrap();
108        let err = CacheManifest::read(dir.path()).expect_err("missing manifest errors");
109        match err {
110            RegistryError::Parse(msg) => assert!(msg.contains("manifest read")),
111            _ => panic!("expected Parse, got {err:?}"),
112        }
113    }
