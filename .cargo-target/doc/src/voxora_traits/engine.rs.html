1//! The [`AsrEngine`] trait and the value types that travel through it.
2//!
3//! An [`AsrEngine`] wraps a single ASR backend (whisper.cpp, qwen3-asr,
4//! future engines) behind a synchronous, `Send + Sync` interface so any
5//! caller can hold an `Arc<dyn AsrEngine>` and swap implementations
6//! without touching inference code.
7
8use crate::error::AsrError;

10/// Reported capabilities of a model.
11///
12/// Engines that do not know their own capabilities should leave every
13/// field at its conservative default (see [`ModelCapabilities::UNKNOWN`]).
14#[derive(Debug, Clone, Default, PartialEq, Eq)]
15#[non_exhaustive]
16pub struct ModelCapabilities {
17    /// Whether the model supports multilingual ASR (as opposed to
18    /// English-only).
19    pub multilingual: bool,
20
21    /// Whether the model can emit word-level timestamps.
22    pub word_timestamps: bool,
23
24    /// Whether the model supports streaming (incremental) inference.
25    pub streaming: bool,
26
27    /// ISO 639-1 language codes the model was trained on. Empty means
28    /// "unknown".
29    pub languages: Vec<String>,
30}
31
32impl ModelCapabilities {
33    /// Sentinel value meaning "capabilities unknown / not advertised".
34    ///
35    /// Returned by the default [`AsrEngine::capabilities`] so implementors
36    /// only override what they actually know.
37    pub const UNKNOWN: Self = Self {
38        multilingual: false,
39        word_timestamps: false,
40        streaming: false,
41        languages: Vec::new(),
42    };
43
44    /// Construct a `ModelCapabilities` from all four fields.
45    ///
46    /// `ModelCapabilities` is `#[non_exhaustive]`, so downstream crates
47    /// cannot use a struct expression. This constructor is the canonical
48    /// way to build one with known values.
49    pub const fn new(
50        multilingual: bool,
51        word_timestamps: bool,
52        streaming: bool,
53        languages: Vec<String>,
54    ) -> Self {
55        Self {
56            multilingual,
57            word_timestamps,
58            streaming,
59            languages,
60        }
61    }
62}
63
64impl std::fmt::Display for ModelCapabilities {
65    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
66        write!(
67            f,
68            "multilingual={}, word_timestamps={}, streaming={}, languages={}",
69            self.multilingual,
70            self.word_timestamps,
71            self.streaming,
72            self.languages.len()
73        )
74    }
75}

77/// Options that affect a single transcription pass but not model selection.
78#[derive(Debug, Clone, Default, PartialEq, Eq)]
79#[non_exhaustive]
80pub struct TranscribeOptions {
81    /// ISO 639-1 code (e.g. `"en"`, `"es"`) or `None` to let the engine
82    /// auto-detect.
83    pub language: Option<String>,
84
85    /// Whether to translate the output to English (only meaningful for
86    /// multilingual models; ignored otherwise).
87    pub translate: bool,
88
89    /// Whether to include per-segment timestamps in
90    /// [`TranscriptionResult::segments`].
91    pub timestamps: bool,
92}
93
94impl TranscribeOptions {
95    /// Construct a [`TranscribeOptions`] from its three fields.
96    ///
97    /// Provided because [`TranscribeOptions`] is `#[non_exhaustive]`
98    /// and so cannot be built with a struct expression outside this
99    /// crate; downstream engines and tests use this constructor to
100    /// pick a non-default combination of options.
101    pub const fn new(language: Option<String>, translate: bool, timestamps: bool) -> Self {
102        Self {
103            language,
104            translate,
105            timestamps,
106        }
107    }
108}

110/// One segment of a transcription.
111#[derive(Debug, Clone, PartialEq, Eq)]
112#[non_exhaustive]
113pub struct TranscriptionSegment {
114    /// Inclusive segment start, in samples at the engine's expected
115    /// sample rate (typically 16 kHz).
116    pub start_sample: u64,
117
118    /// Exclusive segment end, in samples at the engine's expected
119    /// sample rate.
120    pub end_sample: u64,
121
122    /// Transcribed text for this segment.
123    pub text: String,
124}
125
126impl TranscriptionSegment {
127    /// Construct a [`TranscriptionSegment`] from its three fields.
128    ///
129    /// Provided because [`TranscriptionSegment`] is `#[non_exhaustive]`
130    /// and so cannot be built with a struct expression outside this
131    /// crate; engines use this constructor to push one segment per
132    /// decoded window.
133    pub fn new(start_sample: u64, end_sample: u64, text: impl Into<String>) -> Self {
134        Self {
135            start_sample,
136            end_sample,
137            text: text.into(),
138        }
139    }
140}

142/// The full output of a transcription pass.
143#[derive(Debug, Clone, PartialEq, Eq, Default)]
144#[non_exhaustive]
145pub struct TranscriptionResult {
146    /// Full transcribed text, segments joined with a single space.
147    pub text: String,
148
149    /// Language that was forced or detected, if known.
150    pub language: Option<String>,
151
152    /// Per-segment breakdown. Empty when
153    /// [`TranscribeOptions::timestamps`] was `false`.
154    pub segments: Vec<TranscriptionSegment>,
155}
156
157impl TranscriptionResult {
158    /// Construct an empty result with the given full text.
159    ///
160    /// Provided because [`TranscriptionResult`] is `#[non_exhaustive]`
161    /// and so cannot be built with a struct expression outside of this
162    /// crate; this constructor gives engines a one-line way to return a
163    /// minimal result.
164    pub fn new(text: impl Into<String>) -> Self {
165        Self {
166            text: text.into(),
167            language: None,
168            segments: Vec::new(),
169        }
170    }
171
172    /// Construct a result with full text, optional language, and a
173    /// per-segment breakdown.
174    ///
175    /// Used by engines that emit both the joined text and a
176    /// structured segment list (e.g. `voxora-whisper` when the caller
177    /// requested timestamps).
178    pub fn with_segments(
179        text: impl Into<String>,
180        language: Option<String>,
181        segments: Vec<TranscriptionSegment>,
182    ) -> Self {
183        Self {
184            text: text.into(),
185            language,
186            segments,
187        }
188    }
189}
190
191impl std::fmt::Display for TranscriptionResult {
192    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
193        f.write_str(&self.text)
194    }
195}

197/// An automatic-speech-recognition engine.
198///
199/// Inference is exposed synchronously because it is a CPU/GPU-bound
200/// task; wrapping it in `async` would add cost without benefit. Model
201/// acquisition (downloads, cache lookups) lives in
202/// [`crate::ModelSource`] instead.
203pub trait AsrEngine: Send + Sync {
204    /// What this engine supports. Defaults to [`ModelCapabilities::UNKNOWN`]
205    /// so implementors only override what they know.
206    fn capabilities(&self) -> ModelCapabilities {
207        ModelCapabilities::UNKNOWN
208    }
209
210    /// Transcribe a buffer of mono PCM samples at the engine's expected
211    /// sample rate (typically 16 kHz, f32 in `[-1.0, 1.0]`).
212    fn transcribe(
213        &self,
214        samples: &[f32],
215        opts: &TranscribeOptions,
216    ) -> Result<TranscriptionResult, AsrError>;
217}

223    /// Minimal `AsrEngine` used to verify trait-object construction.
224    struct EchoEngine;
225
226    impl AsrEngine for EchoEngine {
227        fn capabilities(&self) -> ModelCapabilities {
228            ModelCapabilities {
229                multilingual: true,
230                languages: vec!["en".into(), "es".into()],
231                ..ModelCapabilities::UNKNOWN
232            }
233        }
234
235        fn transcribe(
236            &self,
237            samples: &[f32],
238            opts: &TranscribeOptions,
239        ) -> Result<TranscriptionResult, AsrError> {
240            Ok(TranscriptionResult {
241                text: format!("echo of {} samples", samples.len()),
242                language: opts.language.clone(),
243                segments: if opts.timestamps && !samples.is_empty() {
244                    vec![TranscriptionSegment {
245                        start_sample: 0,
246                        end_sample: samples.len() as u64,
247                        text: format!("echo of {} samples", samples.len()),
248                    }]
249                } else {
250                    Vec::new()
251                },
252            })
253        }
254    }

264    #[test]
265    fn unknown_capabilities_sentinel_is_distinct_from_default() {
266        // UNKNOWN and Default should both be conservative but they are
267        // reached by different paths — UNKNOWN is what engines get when
268        // they skip overriding capabilities(); Default::default() is for
269        // ad-hoc construction. They must compare equal today but the
270        // contract allows UNKNOWN to diverge later.
271        assert_eq!(ModelCapabilities::UNKNOWN, ModelCapabilities::default());
272    }

289    #[test]
290    fn engine_works_behind_arc_dyn() {
291        let engine: std::sync::Arc<dyn AsrEngine> = std::sync::Arc::new(EchoEngine);
292        let caps = engine.capabilities();
293        assert!(caps.multilingual);

319    #[test]
320    fn engine_works_across_threads() {
321        use std::sync::Arc;
322        use std::thread;
323
324        let engine: Arc<dyn AsrEngine> = Arc::new(EchoEngine);
325        let handles: Vec<_> = (0..4)
326            .map(|i| {
327                let e = Arc::clone(&engine);
328                thread::spawn(move || {
329                    e.transcribe(&vec![0.0_f32; i], &TranscribeOptions::default())
330                        .map(|r| r.text)
331                })
332            })
333            .collect();
334
335        for (i, h) in handles.into_iter().enumerate() {
336            let text = h.join().expect("thread did not panic").expect("ok");
337            assert_eq!(text, format!("echo of {i} samples"));
338        }
339    }
